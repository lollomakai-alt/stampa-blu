Funzione per stampare in base al tipo (Comanda o Conto) async function sendToPrinter(payloadData) { if (!myPrinterDevice || !myPrinterDevice.gatt.connected) { throw new Error("Stampante Bluetooth disconnessa."); } const tipoDoc = (payloadData.type || payloadData.tipo || 'comanda').toLowerCase(); let textToPrint = "\x1B\x40"; // Inizializza stampante if (tipoDoc === 'conto' || tipoDoc === 'preconto') { // ==================== LAYOUT CONTO / PRECONTO ==================== textToPrint += "\x1B\x61\x01"; // Centrato const nomeLocale = payloadData.nome_locale || "NAKAI TIKI BAR"; const indirizzoLocale = payloadData.indirizzo_locale || "Via dei Tiki, 1"; textToPrint += "\x1D\x21\x01"; // Testo grande textToPrint += `${nomeLocale}\n`; textToPrint += "\x1D\x21\x00"; // Torna normale textToPrint += `${indirizzoLocale}\n`; textToPrint += "--------------------------------\n"; textToPrint += "\x1B\x61\x00"; // A sinistra const tavolo = payloadData.tavolo || payloadData.table || "N/D"; textToPrint += `Tavolo: ${tavolo}\n`; textToPrint += `Data: ${new Date().toLocaleDateString()} ${new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })}\n`; textToPrint += "--------------------------------\n"; // Articoli con prezzi if (payloadData.items && Array.isArray(payloadData.items)) { payloadData.items.forEach(item => { const qty = item.qty || 1; const name = item.name || item.nome || 'Articolo'; const prezzo = item.prezzo || item.price; textToPrint += `${qty} x ${name}`; if (prezzo) { textToPrint += ` - ${(qty * prezzo).toFixed(2)}€`; } textToPrint += "\n"; }); } textToPrint += "--------------------------------\n"; // Totale const totale = payloadData.totale || payloadData.total; if (totale) { textToPrint += "\x1B\x61\x02"; // A destra textToPrint += "\x1D\x21\x01"; // Grande textToPrint += `TOTALE: ${Number(totale).toFixed(2)}€\n`; textToPrint += "\x1D\x21\x00"; } textToPrint += "\x1B\x61\x01"; // Centrato per i saluti textToPrint += "\nGrazie per la visita!\nArrivederci!\n\n\n"; } else { // ==================== LAYOUT COMANDA (Cucina/Bar) ==================== textToPrint += "\x1B\x61\x01"; // Centrato // Tavolo in grande textToPrint += "\x1D\x21\x11"; const tavolo = payloadData.tavolo || payloadData.table || "N/D"; textToPrint += `TAVOLO: ${tavolo}\n`; textToPrint += "\x1D\x21\x00"; // Normale textToPrint += "\x1B\x61\x00"; // A sinistra textToPrint += "--------------------------------\n"; // Articoli in grassetto textToPrint += "\x1B\x45\x1"; if (payloadData.items && Array.isArray(payloadData.items)) { payloadData.items.forEach(item => { const qty = item.qty || 1; const name = item.name || item.nome || 'Generico'; textToPrint += `${qty} x ${name}\n`; }); } textToPrint += "\x1B\x45\x0"; // Disattiva grassetto textToPrint += "--------------------------------\n\n\n"; } textToPrint += "\x1B\x69"; // Taglio carta const encoder = new TextEncoder(); const data = encoder.encode(textToPrint); if (bluetoothCharacteristic) { await bluetoothCharacteristic.writeValue(data); } else { throw new Error("Caratteristica di scrittura Bluetooth non trovata."); } }
