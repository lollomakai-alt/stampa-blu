import { createClient } from '@supabase/supabase-js' // Configurazione Supabase (Inserisci i tuoi dati) const SUPABASE_URL = 'IL_TUO_SUPABASE_URL'; const SUPABASE_KEY = 'IL_TUO_SUPABASE_ANON_KEY'; const supabase = createClient(SUPABASE_URL, SUPABASE_KEY); let stampanteDisponibile = null; // Riferimento alla stampante Bluetooth connessa let ultimaComandaElaborataId = null; // Anti-duplicazione stampa // ========================================== // 1. GESTIONE CONNESSIONE BLUETOOTH // ========================================== async function associaStampante() { try { console.log("[SYSTEM] Ricerca stampante termica Bluetooth in corso..."); stampanteDisponibile = await navigator.bluetooth.requestDevice({ acceptAllDevices: true, optionalServices: ['000018f0-0000-1000-8000-00805f9b34fb'] // Servizio standard stampanti termiche ESC/POS }); console.log(`[SYSTEM] Stampante connessa: ${stampanteDisponibile.name}`); document.getElementById("status-stampante").innerText = "Test Stampante (Connessa)"; } catch (error) { console.error("[ERRORE] Connessione Bluetooth fallita:", error); } } // ========================================== // 2. ASCOLTO TEMPO REALE SUPABASE (Realtime) // ========================================== export function avviaAscoltoOrdini() { console.log("[SYSTEM] Avvio ascolto in tempo reale da Supabase..."); supabase .channel('ordini-bridge-lollo') .on( 'postgres_changes', { event: 'INSERT', schema: 'public', table: 'ordini' }, // Sostituisci 'ordini' con la tua tabella async (payload) => { const ordine = payload.new; // Prevenzione stampa doppia per lo stesso ID evento if (ultimaComandaElaborataId === ordine.id) { console.log("[SYSTEM] Ignorato evento duplicato per lo stesso ordine."); return; } ultimaComandaElaborataId = ordine.id; console.log("[SYSTEM] Nuovo record ricevuto in tempo reale!", ordine); // Indirizza al binario corretto in base al tipo (comanda o preconto) if (ordine.tipo === 'preconto') { await stampaPreconto(ordine); } else { await stampaComanda(ordine); } } ) .subscribe(); } // ========================================== // 3. FORMATTAZIONE E STAMPA COMANDA (Cucina) // ========================================== async function stampaComanda(dati) { console.log("[STAMPA] Generazione Comanda (Grande e Visibile)..."); let testoStampa = ""; testoStampa += "\x1B\x45\x01"; // Grassetto ON testoStampa += " COMANDA\n"; testoStampa += `Data: ${dati.data} - Coperti: ${dati.coperti}\n`; testoStampa += "--------------------------------\n"; testoStampa += "DESCRIZIONE QTA\n"; testoStampa += "--------------------------------\n"; // Ciclo articoli con correzione quantitativa da 1.0 a x1 dati.articoli.forEach(item => { let qtaFormattata = `x${parseInt(item.quantita)}`; // Converte es. 1.0 in x1 testoStampa += `${item.nome.padEnd(25)} ${qtaFormattata}\n`; }); testoStampa += "--------------------------------\n"; testoStampa += `Oper: ${dati.operatore}\n\n\n`; testoStampa += "\x1B\x45\x00"; // Grassetto OFF await inviaAlFlussoBluetooth(testoStampa); } // ========================================== // 4. FORMATTAZIONE E STAMPA PRECONTO (Cassa) // ========================================== async function stampaPreconto(dati) { console.log("[STAMPA] Generazione Preconto (Compatto)..."); let testoStampa = ""; testoStampa += "\x1B\x21\x01"; // Carattere compatto/piccolo testoStampa += `*** ${dati.titolo_locale || 'IL MIO LOCALE'} ***\n`; testoStampa += "---------- PRECONTO ----------\n"; dati.articoli.forEach(item => { let qtaFormattata = `x${parseInt(item.quantita)}`; let prezzoTotale = (item. prezzo * item.quantita).toFixed(2); testoStampa += `${qtaFormattata} ${item.nome} - Eur ${prezzoTotale}\n`; }); testoStampa += "--------------------------------\n"; testoStampa += `TOTALE DA PAGARE: Eur ${dati.totale}\n`; testoStampa += "Documento non fiscale\n\n\n"; await inviaAlFlussoBluetooth(testoStampa); } // ========================================== // 5. INVIO FISICO DEI DATI ALLA STAMPANTE // ========================================== async function inviaAlFlussoBluetooth(datiComando) { if (!stampanteDisponibile || !stampanteDisponibile.gatt.connected) { console.error("[ERRORE] Stampante non connessa via Bluetooth!"); return; } try { const server = await stampanteDisponibile.gatt.connect(); const service = await server.getPrimaryService('000018f0-0000-1000-8000-00805f9b34fb'); const characteristic = await service.getCharacteristic('00002af1-0000-1000-8000-00805f9b34fb'); const encoder = new TextEncoder(); await characteristic.writeValue(encoder.encode(datiComando)); console.log("[SYSTEM] Stampa eseguita con successo."); } catch (err) { console.error("[ERRORE] Impossibile inviare i dati alla stampante:", err); } }
