GESTIONE CONNESSIONE BLUETOOTH // ========================================== export async function associaStampante() { try { console.log("[SYSTEM] Ricerca stampante termica Bluetooth in corso..."); stampanteDisponibile = await navigator.bluetooth.requestDevice({ acceptAllDevices: true, optionalServices: ['000018f0-0000-1000-8000-00805f9b34fb'] // Servizio standard ESC/POS }); console.log(`[SYSTEM] Stampante connessa: ${stampanteDisponibile.name}`); const statusEl = document.getElementById("status-stampante"); if (statusEl) statusEl.innerText = "Test Stampante (Connessa)"; } catch (error) { console.error("[ERRORE] Connessione Bluetooth fallita:", error); } } // ========================================== // 2. ASCOLTO TEMPO REALE SUPABASE (Realtime) // ========================================== export function avviaAscoltoOrdini() { console.log("[SYSTEM] Avvio ascolto in tempo reale da Supabase..."); supabase .channel('ordini-bridge-lollo-definitivo') .on( 'postgres_changes', { event: 'INSERT', schema: 'public', table: 'ordini' }, async (payload) => { const ordine = payload.new; // Prevenzione stampa doppia per lo stesso ID evento if (ultimaComandaElaborataId === ordine.id) { console.log("[SYSTEM] Ignorato evento duplicato per lo stesso ordine."); return; } ultimaComandaElaborataId = ordine.id; console.log("[SYSTEM] Nuovo record ricevuto in tempo reale!", ordine); // Instradamento sul binario corretto (Preconto vs Comanda) if (ordine.tipo === 'preconto') { await stampaPreconto(ordine); } else { await stampaComanda(ordine); } } ) .subscribe(); } // ========================================== // 3. FORMATTAZIONE E STAMPA COMANDA (Cucina - Grande e Visibile) // ========================================== async function stampaComanda(dati) { console.log("[STAMPA] Generazione Comanda..."); let testoStampa = ""; testoStampa += "\x1B\x45\x01"; // Grassetto ON testoStampa += "\x1B\x21\x30"; // Carattere doppio altezza/larghezza (Grande) testoStampa += " COMANDA\n"; testoStampa += "\x1B\x21\x00"; // Ritorno a carattere normale per i dettagli testoStampa += `Data: ${dati.data || ''} - Coperti: ${dati.coperti || ''}\n`; testoStampa += "--------------------------------\n"; testoStampa += "DESCRIZIONE QTA\n"; testoStampa += "--------------------------------\n"; // Ciclo articoli con correzione quantitativa (da 1.0 a x1) if (dati.articoli && Array.isArray(dati.articoli)) { dati.articoli.forEach(item => { let qtaPulita = parseInt(item.quantita) || 1; let qtaFormattata = `x${qtaPulita}`; let nomeArticolo = (item.nome || '').substring(0, 25).padEnd(25); testoStampa += `${nomeArticolo} ${qtaFormattata}\n`; }); } testoStampa += "--------------------------------\n"; testoStampa += `Oper: ${dati.operatore || 'N/D'}\n\n\n`; testoStampa += "\x1B\x45\x00"; // Grassetto OFF await inviaAlFlussoBluetooth(testoStampa); } // ========================================== // 4. FORMATTAZIONE E STAMPA PRECONTO (Cassa - Piccolo e Compatto) // ========================================== async function stampaPreconto(dati) { console.log("[STAMPA] Generazione Preconto..."); let titoloLocale = dati.titolo_locale || 'IL MIO LOCALE'; let testoStampa = ""; testoStampa += "\x1B\x21\x01"; // Carattere compatto/piccolo testoStampa += `*** ${titoloLocale} ***\n`; testoStampa += "---------- PRECONTO ----------\n"; if (dati.articoli && Array.isArray(dati.articoli)) { dati.articoli.forEach(item => { let qtaPulita = parseInt(item.quantita) || 1; let qtaFormattata = `x${qtaPulita}`; let prezzoUnitario = parseFloat(item.prezzo) || 0; let prezzoTotale = (prezzoUnitario * qtaPulita).toFixed(2); testoStampa += `${qtaFormattata} ${item.nome} - Eur ${prezzoTotale}\n`; }); } testoStampa += "--------------------------------\n"; testoStampa += `TOTALE DA PAGARE: Eur ${dati.totale || '0.00'}\n`; testoStampa += "Documento non fiscale\n\n\n"; await inviaAlFlussoBluetooth(testoStampa); } // ========================================== // 5. INVIO FISICO DEI DATI ALLA STAMPANTE BLUETOOTH // ========================================== async function inviaAlFlussoBluetooth(datiComando) { if (!stampanteDisponibile || !stampanteDisponibile.gatt.connected) { console.error("[ERRORE] Stampante non connessa via Bluetooth!"); return; } try { const server = await stampanteDisponibile.gatt.connect(); const service = await server.getPrimaryService('000018f0-0000-1000-8000-00805f9b34fb'); const characteristic = await service.getCharacteristic('00002af1-0000-1000-8000-00805f9b34fb'); const encoder = new TextEncoder(); await characteristic.writeValue(encoder.encode(datiComando)); console.log("[SYSTEM] Stampa eseguita con successo."); } catch (err) { console.error("[ERRORE] Impossibile inviare i dati alla stampante:", err); } }
